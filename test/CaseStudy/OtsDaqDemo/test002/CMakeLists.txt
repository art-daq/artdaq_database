SET ( TEST_NAME "load" )

SET ( TEST_NUMBER "001" )

SET ( TEST_DBPROVIDER "filesystem" )
string(RANDOM LENGTH 5 ALPHABET 0123456789 RAND_NAME)
configure_file ( ${CMAKE_CURRENT_SOURCE_DIR}/test${TEST_NUMBER}.src.json ${CMAKE_CURRENT_BINARY_DIR}/test${TEST_NUMBER}.cmp.${RAND_NAME}.json @ONLY )
# Register test with CTest
add_test(NAME OtsDaqDemo_${TEST_NAME}_${TEST_DBPROVIDER}_${TEST_NUMBER}_t
         COMMAND ${CMAKE_BINARY_DIR}/bin/OtsDaqDemo_t -o ${TEST_NAME} -p ${TEST_DBPROVIDER} -c ${TEST_COLLECTION_NAME} -f json -v ${TEST_VERSION_NAME}_${TEST_NUMBER}   -x test${TEST_NUMBER}.cmp.${RAND_NAME}.json
         WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
set_tests_properties(OtsDaqDemo_${TEST_NAME}_${TEST_DBPROVIDER}_${TEST_NUMBER}_t PROPERTIES
    ENVIRONMENT "ARTDAQ_DATABASE_URI=${ARTDAQ_DATABASE_URI_OTS_FILESYSTEM}")
