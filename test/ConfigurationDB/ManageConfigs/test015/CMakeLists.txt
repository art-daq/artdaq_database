SET ( TEST_NAME "ManageConfigs_addconfig" )

SET ( TEST_COLLECTION_NAME_QUAL "addconfig" )

SET ( TEST_NUMBER "015" )

SET ( TEST_FILE_NUMBER "011" )
SET ( TEST_DBPROVIDER "filesystem" )
string(RANDOM LENGTH 5 ALPHABET 0123456789 RAND_NAME)
configure_file ( ${CMAKE_CURRENT_SOURCE_DIR}/test${TEST_FILE_NUMBER}.cmp.json ${CMAKE_CURRENT_BINARY_DIR}/test${TEST_FILE_NUMBER}.cmp.${RAND_NAME}.json @ONLY )
configure_file ( ${CMAKE_CURRENT_SOURCE_DIR}/test${TEST_FILE_NUMBER}.flt.json ${CMAKE_CURRENT_BINARY_DIR}/test${TEST_FILE_NUMBER}.flt.${RAND_NAME}.json @ONLY )
# Register test with CTest
add_test(NAME ${TEST_NAME}_${TEST_DBPROVIDER}_${TEST_NUMBER}_t
         COMMAND ${TOP_CMAKE_BINARY_DIR}/bin/ConfigurationDB_t --searchquery=test${TEST_FILE_NUMBER}.flt.${RAND_NAME}.json -o addconfig -x test${TEST_FILE_NUMBER}.cmp.${RAND_NAME}.json
         WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
set_tests_properties(${TEST_NAME}_${TEST_DBPROVIDER}_${TEST_NUMBER}_t PROPERTIES
    ENVIRONMENT "ARTDAQ_DATABASE_URI=${ARTDAQ_DATABASE_URI_FILESYSTEM}")
